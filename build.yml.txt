name: Build WinBoost

on:
  push:
    tags: ["v*"]          # cria uma Release ao enviar uma tag (ex.: v1.0.0)
  workflow_dispatch: {}    # permite rodar manualmente pela aba Actions

permissions:
  contents: write           # necessário para publicar a Release

jobs:
  build:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-go@v5
        with:
          go-version: "1.24"

      - name: Compilar WinBoost.exe
        run: go build -trimpath -ldflags "-H windowsgui -s -w" -o WinBoost.exe .

      - name: Guardar o .exe como artefato do build
        uses: actions/upload-artifact@v4
        with:
          name: WinBoost
          path: WinBoost.exe

      - name: Publicar na Release (quando for uma tag)
        if: startsWith(github.ref, 'refs/tags/')
        uses: softprops/action-gh-release@v2
        with:
          files: WinBoost.exe
